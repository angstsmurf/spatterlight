            function incAtt(att) {
                pts = parseInt($('#points').html());
                flag = (pts > 0);
                if (flag) {
                  n = parseInt($('#' + att).html());
                  $('#' + att).html(n + 1);
                  $('#points').html(pts - 1);
                }
            }

            function decAtt(att) {
                n = parseInt($('#' + att).html());
                flag = (n > 0);
                if (flag) {
                  pts = parseInt($('#points').html());
                  $('#' + att).html(n - 1);
                  $('#points').html(pts + 1);
                }
            }
